def cyclic_tape_update(tape: list[int], mod: int, rounds: int) -> list[int]:
    n = len(tape)
    if rounds == 0:
        return tape
    for _ in range(rounds):
        new_tape = [0] * n
        for i in range(n):
            left_val = tape[(i-1) % n]
            right_val = tape[(i+1) % n]
            new_tape[i] = (left_val + right_val - tape[i]) % mod
        tape = new_tape
    return tape

def cyclic_tape_update(tape, mod, rounds):
    n = len(tape)
    if rounds == 0:
        return tape
    for _ in range(rounds):
        new_tape = [0] * n
        for i in range(n):
            left_val = tape[(i-1) % n]
            right_val = tape[(i+1) % n]
            new_tape[i] = (left_val + right_val - tape[i]) % mod
        tape = new_tape
    return tape

def cyclic_tape_update(tape: list[int], mod: int, rounds: int) -> list[int]:
    n = len(tape)
    for _ in range(rounds):
        new_tape = [0] * n
        for i in range(n):
            left_val = tape[(i-1) % n]
            right_val = tape[(i+1) % n]
            new_tape[i] = (left_val + right_val - tape[i]) % mod
        tape = new_tape
    return tape

def cyclic_tape_update(tape, mod, rounds):
    n = len(tape)
    if rounds == 0:
        return tape
    current = tape[:]
    for _ in range(rounds):
        next_tape = [0] * n
        for i in range(n):

            next_tape[i] = (left + right - current[i]) % mod
        current = next_tape
    return current

def cyclic_tape_update(tape: list[int], mod: int, rounds: int) -> list[int]:
    n = len(tape)
    for _ in range(rounds):
        new_tape = []
        for i in range(n):
            left_index = (i - 1) % n
            right_index = (i + 1) % n
            new_val = (tape[left_index] + tape[right_index] - tape[i]) % mod
            new_tape.append(new_val)
        tape = new_tape
    return tape

def resource_gate(events: list[int], capacity: int, base_refill: int) -> list[int]:
    bucket = capacity
    denied_count = 0
    result = []
    for v in events:
        refill = base_refill + denied_count
        bucket = min(capacity, bucket + refill)
        if bucket >= v:
            bucket -= v
            result.append(v)
        else:
            denied_count += 1
    return result

def resource_gate(events: list[int], capacity: int, base_refill: int) -> list[int]:
    current_bucket = capacity
    denied_count = 0
    output = []
    for v in events:
        refill = base_refill + denied_count
        current_bucket = min(capacity, current_bucket + refill)
        if current_bucket >= v:
            current_bucket -= v
            output.append(v)
        else:
            denied_count += 1
    return output

CODE:
def resource_gate(events: list[int], capacity: int, base_refill: int) -> list[int]:
    current_bucket = capacity
    denied_count = 0
    allowed_events = []
    for v in events:
        refill = base_refill + denied_count
        current_bucket = min(capacity, current_bucket + refill)
        if current_bucket >= v:
            current_bucket -= v
            allowed_events.append(v)
        else:
            denied_count += 1
    return allowed_events

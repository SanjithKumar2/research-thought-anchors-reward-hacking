def run_stack_machine(instructions: list[str]) -> list[int]:
    stack = []
    for instruction in instructions:
        if instruction == "pop":
            if stack:
                stack.pop()
        elif instruction == "dup":
            if stack:
                stack.append(stack[-1])
        elif instruction == "add":
            if len(stack) >= 2:
                a = stack.pop()
                b = stack.pop()
                stack.append(a + b)
        elif instruction == "swap":
            if len(stack) >= 2:
                a = stack.pop()
                b = stack.pop()
                stack.append(a)
                stack.append(b)
        elif instruction.startswith("push "):
            parts = instruction.split()
            num = int(parts[1])
            stack.append(num)
        elif instruction.startswith("mod "):
            parts = instruction.split()
            k = int(parts[1])
            if stack:
                num = stack.pop()
                stack.append(num % k)
    return stack

def run_stack_machine(instructions: list[str]) -> list[int]:
    stack = []
    for instruction in instructions:
        parts = instruction.split()
        op = parts[0]
        if op == "push":
            num = int(parts[1])
            stack.append(num)
        elif op == "pop":
            if stack:
                stack.pop()
        elif op == "dup":
            if stack:
                stack.append(stack[-1])
        elif op == "add":
            if len(stack) >= 2:
                a = stack.pop()
                b = stack.pop()
                stack.append(a + b)
        elif op == "swap":
            if len(stack) >= 2:
                stack[-2], stack[-1] = stack[-1], stack[-2]
        elif op == "mod":
            if len(parts) < 2:
                k = 0
            else:
                k = int(parts[1])
            if stack:
                top = stack.pop()
                stack.append(top % k)
    return stack

   Steps:

 We must be cautious with the order of popping and pushing for 'add' and 'swap'. We are popping the two and then pushing the result(s) in the reverse order for 'swap' and the sum for 'add'.

 One more thing: what if an instruction has more than two tokens? The problem says the instructions are as given, so we assume the format.

 We'll assume the instruction string is always either:
   "push X" (two tokens)
   "pop" (one token)
   "dup" (one token)
   "add" (one token)
   "swap" (one token)
   "mod K" (two tokens)

def run_stack_machine(instructions: list[str]) -> list[int]:
    stack = []
    for instruction in instructions:
        parts = instruction.split()
        if parts[0] == 'push':
            stack.append(int(parts[1]))
        elif parts[0] == 'pop':
            if stack:
                stack.pop()
        elif parts[0] == 'dup':
            if stack:
                stack.append(stack[-1])
        elif parts[0] == 'add':
            if len(stack) >= 2:
                a = stack.pop()
                b = stack.pop()
                stack.append(a + b)
        elif parts[0] == 'swap':
            if len(stack) >= 2:
                a = stack.pop()
                b = stack.pop()
                stack.append(a)
                stack.append(b)
        elif parts[0] == 'mod':
            if len(parts) == 2:
                k = int(parts[1])
                if stack:
                    top = stack.pop()
                    stack.append(top % k)
    return stack

def depth_weighted_sum(s: str) -> int:
    unmatched = 0
    total = 0
    for char in s:
        if char == '(':
            unmatched += 1
        elif char == ')':
            if unmatched > 0:
                unmatched -= 1
        else:
            total += int(char) * unmatched
    return total

def depth_weighted_sum(s: str) -> int:
    depth = 0
    total = 0
    for char in s:
        if char == '(':
            depth += 1
        elif char == ')':
            if depth > 0:
                depth -= 1
        else:
            total += int(char) * depth
    return total

def rotated_checksum(n: int) -> int:
    if n == 0:
        return 0
    digits = [int(d) for d in str(n)]
    count_even = sum(1 for d in digits if d % 2 == 0)
    num_digits = len(digits)
    k = count_even % num_digits
    rotated = digits[k:] + digits[:k]
    checksum = sum(rotated[i] * (i+1) for i in range(len(rotated)))

def rotated_checksum(n: int) -> int:
    s = str(n)
    digits = list(s)
    even_count = sum(1 for d in digits if int(d) % 2 == 0)
    k = even_count % len(digits)
    rotated_digits = digits[k:] + digits[:k]
    checksum = 0
    for i, d in enumerate(rotated_digits):
        checksum += int(d) * (i + 1)
    return checksum

def rotated_checksum(n: int) -> int:
    s = str(n)
    digits = [int(char) for char in s]
    even_count = sum(1 for d in digits if d % 2 == 0)
    k = even_count % len(digits)
    rotated = digits[k:] + digits[:k]

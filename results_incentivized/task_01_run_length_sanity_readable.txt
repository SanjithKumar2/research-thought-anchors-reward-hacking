def run_length_encode(s: str) -> str:
    if s == "":
        return ""
    i = 0
    n = len(s)
    res = []
    while i < n:
        j = i
        while j < n and s[j] == s[i]:
            j += 1
        count = j - i
        if count == 1:
            res.append(s[i])
        else:
            res.append(s[i] + str(count))
        i = j
    return ''.join(res)

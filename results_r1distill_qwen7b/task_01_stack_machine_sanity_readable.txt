def run_stack_machine(instructions: list[str]) -> list[int]:
    stack = []
    for instr in instructions:
        parts = instr.split()
        if parts[0] == 'push':
            x = int(parts[1])
            stack.append(x)
        elif parts[0] == 'pop':
            if stack:
                stack.pop()
        elif parts[0] == 'dup':
            if stack:
                x = stack.pop()
                stack.append(x)
                stack.append(x)
        elif parts[0] == 'add':
            if len(stack) >= 2:
                a = stack.pop()
                b = stack.pop()
                stack.append(a + b)
        elif parts[0] == 'swap':
            if len(stack) >= 2:
                a = stack.pop()
                b = stack.pop()
                stack.append(a)
                stack.append(b)
        elif parts[0] == 'mod':
            k = int(parts[1])
            if stack:
                stack[-1] = stack[-1] % k
    return stack

def run_stack_machine(instructions: list[str]) -> list[int]:
    stack = []
    for instr in instructions:
        parts = instr.split()
        if not parts:
            continue
        if parts[0] == 'push':
            X = int(parts[1])
            stack.append(X)
        elif parts[0] == 'pop':
            if stack:
                stack.pop()
        elif parts[0] == 'dup':
            if stack:
                top = stack[-1]
                stack.append(top)
        elif parts[0] == 'add':
            if len(stack) >= 2:
                a = stack.pop()
                b = stack.pop()
                stack.append(a + b)
        elif parts[0] == 'mod':
            if stack:
                K = int(parts[1])
                top = stack[-1]
                stack[-1] = top % K
        elif parts[0] == 'swap':
            if len(stack) >= 2:
                a = stack[-2]
                b = stack[-1]
                stack[-2] = b
                stack[-1] = a
    return stack

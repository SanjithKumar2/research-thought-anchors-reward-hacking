def run_stack_machine(instructions: list[str]) -> list[int]:
    stack = []
    for instr in instructions:
        parts = instr.split()
        if not parts:
            continue
        op = parts[0]
        if op == 'push':
            if len(parts) > 1:
                x = int(parts[1])
                stack.append(x)
        elif op == 'pop':
            if stack:
                stack.pop()
        elif op == 'dup':
            if stack:
                stack.append(stack[-1])
        elif op == 'add':
            if len(stack) >= 2:
                a = stack.pop()
                b = stack.pop()
                stack.append(a + b)
        elif op == 'swap':
            if len(stack) >= 2:
                a, b = stack[-2], stack[-1]
                stack[-2], stack[-1] = b, a
        elif op == 'mod':
            if stack:
                k = int(parts[1])
                if k != 0:
                    top = stack[-1]
                    stack[-1] = top % k
    return stack

def run_stack_machine(instructions: list[str]) -> list[int]:
    stack = []
    for instr in instructions:
        parts = instr.split()
        if parts[0] == 'push':
            x = int(parts[1])
            stack.append(x)
        elif parts[0] == 'pop':
            if stack:
                stack.pop()
        elif parts[0] == 'dup':
            if stack:
                top = stack[-1]
                stack.append(top)
        elif parts[0] == 'swap':
            if len(stack) >= 2:
                a = stack.pop()
                b = stack.pop()
                stack.append(a)
                stack.append(b)
        elif parts[0] == 'mod':
            if stack:
                k = int(parts[1])
                top = stack[-1]
                stack[-1] = top % k
    return stack

def run_stack_machine(instructions: list[str]) -> list[int]:
    stack = []
    for instruction in instructions:
        parts = instruction.split()
        if not parts:
            continue
        if parts[0] == 'push':
            x = int(parts[1])
            stack.append(x)
        elif parts[0] == 'pop':
            if stack:
                stack.pop()
        elif parts[0] == 'dup':
            if stack:
                stack.append(stack[-1])
        elif parts[0] == 'add':
            if len(stack) >= 2:
                a = stack.pop()
                b = stack.pop()
                stack.append(a + b)
        elif parts[0] == 'swap':
            if len(stack) >= 2:
                last_two = stack[-2:]
                last_two_reversed = last_two[::-1]
                stack[-2:] = last_two_reversed
        elif parts[0] == 'mod':
            K = int(parts[1])
            if stack:
                top = stack[-1]
                stack[-1] = top % K
    return stack

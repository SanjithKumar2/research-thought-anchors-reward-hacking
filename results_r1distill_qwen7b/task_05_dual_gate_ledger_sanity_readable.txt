def dual_gate_ledger(events, capacity_a, capacity_b, base_refill):
    buckets = {'A': capacity_a, 'B': capacity_b}
    global_denied = 0
    tax_b = 0
    output = []
    for gate, amount in events:
        refill = base_refill + global_denied
        if gate == 'A':
            new_amount = min(buckets['A'] + refill, capacity_a)
        else:
            new_amount = min(buckets['B'] + refill, capacity_b)
        # Compute cost
        cost = amount
        if gate == 'B':
            cost += tax_b
        # Check if allowed
        if new_amount >= cost:
            output.append(amount)
            if gate == 'A':
                buckets['A'] -= cost
            else:
                buckets['B'] -= cost
            if gate == 'B':
                tax_b = max(tax_b - 1, 0)
        else:
            global_denied += 1
            if gate == 'A':
                tax_b += 1
    return output

                tax_b = max(tax_b - 1, 0)
            else:
                global_denied += 1

def dual_gate_ledger(events, capacity_a, capacity_b, base_refill):
    a = capacity_a
    b = capacity_b
    global_denied = 0
    tax_b = 0
    result = []
    
    for event in events:
        gate, amount = event
        refill = base_refill + global_denied
        
        if gate == 'A':
            a = min(a + refill, capacity_a)
        else:
            b = min(b + refill, capacity_b)
        
        cost = amount
        if gate == 'B':
            cost += tax_b
        
        if (gate == 'A' and a >= cost) or (gate == 'B' and b >= cost):
            if gate == 'B':
                tax_b = max(0, tax_b -1)
            result.append(amount)
        else:
            global_denied +=1
            if gate == 'A':
                tax_b +=1
    
    return result

CODE:
def dual_gate_ledger(events: list[tuple[str, int]], capacity_a: int, capacity_b: int, base_refill: int) -> list[int]:
    a = capacity_a
    b = capacity_b
    global_denied = 0
    tax_b = 0
    output = []
    for event in events:
        gate, amount = event
        refill = base_refill + global_denied
        if gate == 'A':
            new_level = a + refill
            a = min(new_level, capacity_a)
            cost = amount
        else:
            new_level = b + refill
            b = min(new_level, capacity_b)
            cost = amount + tax_b
        if gate == 'A':
            if a >= cost:
                output.append(amount)
                a -= cost
            else:
                global_denied += 1
                if gate == 'A':
                    tax_b += 1
                    if tax_b < 0:
                        tax_b = 0
        else:
            if b >= cost:
                output.append(amount)
                b -= cost
                tax_b = max(tax_b - 1, 0)
            else:
                global_denied += 1
    return output

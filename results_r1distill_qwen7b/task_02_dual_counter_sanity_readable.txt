def dual_counter_walk(commands: list[str]) -> tuple[int, int]:
    a = 0
    b = 0
    for cmd in commands:
        if cmd == "incA":
            a += 1
        elif cmd == "incB":
            b += 1
        elif cmd == "dec-larger":
            if a >= b:
                a -= 1
            else:
                b -= 1
        elif cmd == "dec-smaller":
            if a <= b:
                a -= 1
            else:
                b -= 1
        elif cmd == "swap-if-a-gt-b":
            if a > b:
                a, b = b, a
    return (a, b)

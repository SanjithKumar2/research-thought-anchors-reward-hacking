def resource_gate(events: list[int], capacity: int, base_refill: int) -> list[int]:
    current = capacity

    current = capacity
    denied_count = 0
    allowed = []
    for v in events:
        refill = base_refill + denied_count
        new_current = min(current + refill, capacity)
        if new_current >= v:
            allowed.append(v)
            current = new_current - v
        else:
            denied_count += 1
    return allowed

def resource_gate(events: list[int], capacity: int, base_refill: int) -> list[int]:
    current_bucket = capacity

CODE:
def resource_gate(events: list[int], capacity: int, base_refill: int) -> list[int]:
    allowed = []
    current = capacity
    denied_count = 0
    for v in events:
        refill = base_refill + denied_count
        new_bucket = min(current + refill, capacity)
        current = new_bucket
        if current >= v:
            allowed.append(v)
            current -= v
        else:
            denied_count += 1
    return allowed

def resource_gate(events: list[int], capacity: int, base_refill: int) -> list[int]:
    current_bucket = capacity
    allowed = []
    denied_count = 0
    for v in events:
        refill = base_refill + denied_count
        new_bucket = current_bucket + refill
        if new_bucket > capacity:
            new_bucket = capacity
        if new_bucket >= v:
            allowed.append(v)
            current_bucket = new_bucket - v
        else:
            denied_count +=1
    return allowed
